import { useState } from "react";

const navlinks = [
  { name: "dashboard", path: "/dashboard" },

  {
    name: "student",
    children: [
      { name: "add user", path: "/adduser" },
      { name: "users", path: "/users" },
    ],
  },

  { name: "setting", path: "/setting" },
];

function Sidebar() {
  const [openTab, setOpenTab] = useState("");

  return (
    <div>
      {navlinks.map((link) => (
        <div key={link.name}>
          
          <div
            onClick={() => {
              if (link.children) {
                setOpenTab(openTab === link.name ? "" : link.name);
              } else {
                navigate(link.path);
              }
            }}
          >
            {link.name}
          </div>

          {openTab === link.name &&
            link.children &&
            link.children.map((child) => (
              <div
                key={child.name}
                onClick={() => navigate(child.path)}
              >
                {child.name}
              </div>
            ))}
            
        </div>
      ))}
    </div>
  );
}
